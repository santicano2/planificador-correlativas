# Dark Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** Add a persistent light/dark theme toggle with a cold graphite dark palette and a header symbol.

**Architecture:** Convert the Tailwind color tokens to runtime CSS variables, define dark overrides under `html[data-theme="dark"]`, and use a small inline script plus a header toggle to set and persist the theme. A pure helper resolves stored/system preferences so the behavior is testable.

**Tech Stack:** Next.js 16 App Router, React 19, Tailwind CSS 4, TypeScript, Vitest.

```ts
export type Theme = "light" | "dark";

export const THEME_STORAGE_KEY = "planificador-correlativas:theme:v1";

export function isTheme(value: unknown): value is Theme {
  return value === "light" || value === "dark";
}

export function resolveInitialTheme(
  storedTheme: string | null | undefined,
  prefersDark: boolean,
): Theme {
  if (isTheme(storedTheme)) {
    return storedTheme;
  }

  return prefersDark ? "dark" : "light";
}

export function getNextTheme(theme: Theme): Theme {
  return theme === "dark" ? "light" : "dark";
}
```

- [x] **Step 2: Create the theme toggle**

Create `src/components/theme/theme-toggle.tsx`:

```tsx
"use client";

import { useEffect } from "react";

import { MoonIcon, SunIcon } from "@/components/icons";
import {
  THEME_STORAGE_KEY,
  getNextTheme,
  isTheme,
  resolveInitialTheme,
  type Theme,
} from "@/lib/theme/theme";

let hasExplicitChoice = false;

function applyTheme(theme: Theme): void {
  const root = document.documentElement;
  root.dataset.theme = theme;
  root.style.colorScheme = theme;
}

function readStoredTheme(): string | null {
  try {
    return window.localStorage.getItem(THEME_STORAGE_KEY);
  } catch {
    return null;
  }
}

function writeStoredTheme(theme: Theme): void {
  try {
    window.localStorage.setItem(THEME_STORAGE_KEY, theme);
  } catch {}
}

export function ThemeToggle() {
  useEffect(() => {
    if (hasExplicitChoice || isTheme(readStoredTheme())) {
      return;
    }

    const media = window.matchMedia("(prefers-color-scheme: dark)");

    const handleSystemChange = () => {
      if (hasExplicitChoice || isTheme(readStoredTheme())) {
        return;
      }

      applyTheme(resolveInitialTheme(null, media.matches));
    };

    media.addEventListener("change", handleSystemChange);

    return () => media.removeEventListener("change", handleSystemChange);
  }, []);

  const toggleTheme = () => {
    const root = document.documentElement;
    const current = isTheme(root.dataset.theme) ? root.dataset.theme : "light";
    const next = getNextTheme(current);

    hasExplicitChoice = true;
    applyTheme(next);
    writeStoredTheme(next);
  };

  return (
    <button
      type="button"
      onClick={toggleTheme}
      className="btn btn-secondary btn-sm size-8 p-0"
      aria-label="Cambiar tema"
      title="Cambiar tema"
    >
      <MoonIcon className="theme-icon theme-icon--light size-4" />
      <SunIcon className="theme-icon theme-icon--dark size-4" />
    </button>
  );
}
```
